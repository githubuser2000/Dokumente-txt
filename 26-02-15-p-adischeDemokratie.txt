import matplotlib.pyplot as plt
import numpy as np
from fractions import Fraction as F

# 1️⃣ 3-adische Zahl 10/7, 10 Stellen
x = F(10, 7)
p = 3
precision = 10

# 2️⃣ Berechne p-adische Ziffern
def padic_digits(x, p, precision):
    digits = []
    current = x
    for _ in range(precision):
        inv_den = pow(current.denominator, -1, p)
        a_i = (current.numerator * inv_den) % p
        digits.append(a_i)
        current = (current - a_i) / p
    return digits

digits = padic_digits(x, p, precision)
print("3-adische Ziffern:", digits)

# 3️⃣ Kreisdiagramm vorbereiten
fig, ax = plt.subplots(figsize=(6,6), subplot_kw={'polar':True})
ax.set_theta_direction(-1)  # Uhrzeigersinn
ax.set_theta_offset(np.pi/2)  # Start oben
ax.set_xticks([])
ax.set_yticks([])

# 4️⃣ Von innen nach außen Drittelungen
theta_start = 0
theta_end = 2*np.pi
r_inner = 0
r_width = 1  # Radiale Breite pro Ebene

for i, a in enumerate(digits):
    theta_segment = (theta_end - theta_start) / p
    theta_start += a * theta_segment
    theta_end = theta_start + theta_segment

    # Kreisring zeichnen
    theta = np.linspace(theta_start, theta_end, 100)
    r = np.full_like(theta, r_inner + r_width)
    ax.plot(theta, r, color='C0', lw=4)  # Pfad
    r_inner += r_width

# 5️⃣ Endwinkel berechnen
end_angle = theta_start + (theta_end - theta_start)/2
cot_value = 1/np.tan(end_angle)
print(f"Endwinkel (rad): {end_angle:.4f}, cotangent: {cot_value:.4f}")

plt.title("3-adische Navigation von 10/7 auf Kreis")
plt.show()
